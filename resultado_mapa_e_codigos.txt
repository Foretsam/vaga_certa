#!/usr/bin/perl

use strict;
use warnings;
use IPC::Open2;

# An example hook script to integrate Watchman
# (https://facebook.github.io/watchman/) with git to speed up detecting
# new and modified files.
#
# The hook is passed a version (currently 2) and last update token
# formatted as a string and outputs to stdout a new update token and
# all files that have been modified since the update token. Paths must
# be relative to the root of the working tree and separated by a single NUL.
#
# To enable this hook, rename this file to "query-watchman" and set
# 'git config core.fsmonitor .git/hooks/query-watchman'
#
my ($version, $last_update_token) = @ARGV;

# Uncomment for debugging
# print STDERR "$0 $version $last_update_token\n";

# Check the hook interface version
if ($version ne 2) {
	die "Unsupported query-fsmonitor hook version '$version'.\n" .
	    "Falling back to scanning...\n";
}

my $git_work_tree = get_working_dir();

my $retry = 1;

my $json_pkg;
eval {
	require JSON::XS;
	$json_pkg = "JSON::XS";
	1;
} or do {
	require JSON::PP;
	$json_pkg = "JSON::PP";
};

launch_watchman();

sub launch_watchman {
	my $o = watchman_query();
	if (is_work_tree_watched($o)) {
		output_result($o->{clock}, @{$o->{files}});
	}
}

sub output_result {
	my ($clockid, @files) = @_;

	# Uncomment for debugging watchman output
	# open (my $fh, ">", ".git/watchman-output.out");
	# binmode $fh, ":utf8";
	# print $fh "$clockid\n@files\n";
	# close $fh;

	binmode STDOUT, ":utf8";
	print $clockid;
	print "\0";
	local $, = "\0";
	print @files;
}

sub watchman_clock {
	my $response = qx/watchman clock "$git_work_tree"/;
	die "Failed to get clock id on '$git_work_tree'.\n" .
		"Falling back to scanning...\n" if $? != 0;

	return $json_pkg->new->utf8->decode($response);
}

sub watchman_query {
	my $pid = open2(\*CHLD_OUT, \*CHLD_IN, 'watchman -j --no-pretty')
	or die "open2() failed: $!\n" .
	"Falling back to scanning...\n";

	# In the query expression below we're asking for names of files that
	# changed since $last_update_token but not from the .git folder.
	#
	# To accomplish this, we're using the "since" generator to use the
	# recency index to select candidate nodes and "fields" to limit the
	# output to file names only. Then we're using the "expression" term to
	# further constrain the results.
	my $last_update_line = "";
	if (substr($last_update_token, 0, 1) eq "c") {
		$last_update_token = "\"$last_update_token\"";
		$last_update_line = qq[\n"since": $last_update_token,];
	}
	my $query = <<"	END";
		["query", "$git_work_tree", {$last_update_line
			"fields": ["name"],
			"expression": ["not", ["dirname", ".git"]]
		}]
	END

	# Uncomment for debugging the watchman query
	# open (my $fh, ">", ".git/watchman-query.json");
	# print $fh $query;
	# close $fh;

	print CHLD_IN $query;
	close CHLD_IN;
	my $response = do {local $/; <CHLD_OUT>};

	# Uncomment for debugging the watch response
	# open ($fh, ">", ".git/watchman-response.json");
	# print $fh $response;
	# close $fh;

	die "Watchman: command returned no output.\n" .
	"Falling back to scanning...\n" if $response eq "";
	die "Watchman: command returned invalid output: $response\n" .
	"Falling back to scanning...\n" unless $response =~ /^\{/;

	return $json_pkg->new->utf8->decode($response);
}

sub is_work_tree_watched {
	my ($output) = @_;
	my $error = $output->{error};
	if ($retry > 0 and $error and $error =~ m/unable to resolve root .* directory (.*) is not watched/) {
		$retry--;
		my $response = qx/watchman watch "$git_work_tree"/;
		die "Failed to make watchman watch '$git_work_tree'.\n" .
		    "Falling back to scanning...\n" if $? != 0;
		$output = $json_pkg->new->utf8->decode($response);
		$error = $output->{error};
		die "Watchman: $error.\n" .
		"Falling back to scanning...\n" if $error;

		# Uncomment for debugging watchman output
		# open (my $fh, ">", ".git/watchman-output.out");
		# close $fh;

		# Watchman will always return all files on the first query so
		# return the fast "everything is dirty" flag to git and do the
		# Watchman query just to get it over with now so we won't pay
		# the cost in git to look up each individual file.
		my $o = watchman_clock();
		$error = $output->{error};

		die "Watchman: $error.\n" .
		"Falling back to scanning...\n" if $error;

		output_result($o->{clock}, ("/"));
		$last_update_token = $o->{clock};

		eval { launch_watchman() };
		return 0;
	}

	die "Watchman: $error.\n" .
	"Falling back to scanning...\n" if $error;

	return 1;
}

sub get_working_dir {
	my $working_dir;
	if ($^O =~ 'msys' || $^O =~ 'cygwin') {
		$working_dir = Win32::GetCwd();
		$working_dir =~ tr/\\/\//;
	} else {
		require Cwd;
		$working_dir = Cwd::cwd();
	}

	return $working_dir;
}

#!/bin/sh
#
# An example hook script to verify what is about to be committed.
# Called by "git commit" with no arguments.  The hook should
# exit with non-zero status after issuing an appropriate message if
# it wants to stop the commit.
#
# To enable this hook, rename this file to "pre-commit".

if git rev-parse --verify HEAD >/dev/null 2>&1
then
	against=HEAD
else
	# Initial commit: diff against an empty tree object
	against=$(git hash-object -t tree /dev/null)
fi

# If you want to allow non-ASCII filenames set this variable to true.
allownonascii=$(git config --type=bool hooks.allownonascii)

# Redirect output to stderr.
exec 1>&2

# Cross platform projects tend to avoid non-ASCII filenames; prevent
# them from being added to the repository. We exploit the fact that the
# printable range starts at the space character and ends with tilde.
if [ "$allownonascii" != "true" ] &&
	# Note that the use of brackets around a tr range is ok here, (it's
	# even required, for portability to Solaris 10's /usr/bin/tr), since
	# the square bracket bytes happen to fall in the designated range.
	test $(git diff-index --cached --name-only --diff-filter=A -z $against |
	  LC_ALL=C tr -d '[ -~]\0' | wc -c) != 0
then
	cat <<\EOF
Error: Attempt to add a non-ASCII file name.

This can cause problems if you want to work with people on other platforms.

To be portable it is advisable to rename the file.

If you know what you are doing you can disable this check using:

  git config hooks.allownonascii true
EOF
	exit 1
fi

# If there are whitespace errors, print the offending file names and fail.
exec git diff-index --check --cached $against --

#!/bin/sh

# An example hook script to verify what is about to be pushed.  Called by "git
# push" after it has checked the remote status, but before anything has been
# pushed.  If this script exits with a non-zero status nothing will be pushed.
#
# This hook is called with the following parameters:
#
# $1 -- Name of the remote to which the push is being done
# $2 -- URL to which the push is being done
#
# If pushing without using a named remote those arguments will be equal.
#
# Information about the commits which are being pushed is supplied as lines to
# the standard input in the form:
#
#   <local ref> <local oid> <remote ref> <remote oid>
#
# This sample shows how to prevent push of commits where the log message starts
# with "WIP" (work in progress).

remote="$1"
url="$2"

zero=$(git hash-object --stdin </dev/null | tr '[0-9a-f]' '0')

while read local_ref local_oid remote_ref remote_oid
do
	if test "$local_oid" = "$zero"
	then
		# Handle delete
		:
	else
		if test "$remote_oid" = "$zero"
		then
			# New branch, examine all commits
			range="$local_oid"
		else
			# Update to existing branch, examine new commits
			range="$remote_oid..$local_oid"
		fi

		# Check for WIP commit
		commit=$(git rev-list -n 1 --grep '^WIP' "$range")
		if test -n "$commit"
		then
			echo >&2 "Found WIP commit in $local_ref, not pushing"
			exit 1
		fi
	fi
done

exit 0

#!/bin/sh
#
# Copyright (c) 2006, 2008 Junio C Hamano
#
# The "pre-rebase" hook is run just before "git rebase" starts doing
# its job, and can prevent the command from running by exiting with
# non-zero status.
#
# The hook is called with the following parameters:
#
# $1 -- the upstream the series was forked from.
# $2 -- the branch being rebased (or empty when rebasing the current branch).
#
# This sample shows how to prevent topic branches that are already
# merged to 'next' branch from getting rebased, because allowing it
# would result in rebasing already published history.

publish=next
basebranch="$1"
if test "$#" = 2
then
	topic="refs/heads/$2"
else
	topic=`git symbolic-ref HEAD` ||
	exit 0 ;# we do not interrupt rebasing detached HEAD
fi

case "$topic" in
refs/heads/??/*)
	;;
*)
	exit 0 ;# we do not interrupt others.
	;;
esac

# Now we are dealing with a topic branch being rebased
# on top of master.  Is it OK to rebase it?

# Does the topic really exist?
git show-ref -q "$topic" || {
	echo >&2 "No such branch $topic"
	exit 1
}

# Is topic fully merged to master?
not_in_master=`git rev-list --pretty=oneline ^master "$topic"`
if test -z "$not_in_master"
then
	echo >&2 "$topic is fully merged to master; better remove it."
	exit 1 ;# we could allow it, but there is no point.
fi

# Is topic ever merged to next?  If so you should not be rebasing it.
only_next_1=`git rev-list ^master "^$topic" ${publish} | sort`
only_next_2=`git rev-list ^master           ${publish} | sort`
if test "$only_next_1" = "$only_next_2"
then
	not_in_topic=`git rev-list "^$topic" master`
	if test -z "$not_in_topic"
	then
		echo >&2 "$topic is already up to date with master"
		exit 1 ;# we could allow it, but there is no point.
	else
		exit 0
	fi
else
	not_in_next=`git rev-list --pretty=oneline ^${publish} "$topic"`
	/usr/bin/perl -e '
		my $topic = $ARGV[0];
		my $msg = "* $topic has commits already merged to public branch:\n";
		my (%not_in_next) = map {
			/^([0-9a-f]+) /;
			($1 => 1);
		} split(/\n/, $ARGV[1]);
		for my $elem (map {
				/^([0-9a-f]+) (.*)$/;
				[$1 => $2];
			} split(/\n/, $ARGV[2])) {
			if (!exists $not_in_next{$elem->[0]}) {
				if ($msg) {
					print STDERR $msg;
					undef $msg;
				}
				print STDERR " $elem->[1]\n";
			}
		}
	' "$topic" "$not_in_next" "$not_in_master"
	exit 1
fi

<<\DOC_END

This sample hook safeguards topic branches that have been
published from being rewound.

The workflow assumed here is:

 * Once a topic branch forks from "master", "master" is never
   merged into it again (either directly or indirectly).

 * Once a topic branch is fully cooked and merged into "master",
   it is deleted.  If you need to build on top of it to correct
   earlier mistakes, a new topic branch is created by forking at
   the tip of the "master".  This is not strictly necessary, but
   it makes it easier to keep your history simple.

 * Whenever you need to test or publish your changes to topic
   branches, merge them into "next" branch.

The script, being an example, hardcodes the publish branch name
to be "next", but it is trivial to make it configurable via
$GIT_DIR/config mechanism.

With this workflow, you would want to know:

(1) ... if a topic branch has ever been merged to "next".  Young
    topic branches can have stupid mistakes you would rather
    clean up before publishing, and things that have not been
    merged into other branches can be easily rebased without
    affecting other people.  But once it is published, you would
    not want to rewind it.

(2) ... if a topic branch has been fully merged to "master".
    Then you can delete it.  More importantly, you should not
    build on top of it -- other people may already want to
    change things related to the topic as patches against your
    "master", so if you need further changes, it is better to
    fork the topic (perhaps with the same name) afresh from the
    tip of "master".

Let's look at this example:

		   o---o---o---o---o---o---o---o---o---o "next"
		  /       /           /           /
		 /   a---a---b A     /           /
		/   /               /           /
	       /   /   c---c---c---c B         /
	      /   /   /             \         /
	     /   /   /   b---b C     \       /
	    /   /   /   /             \     /
    ---o---o---o---o---o---o---o---o---o---o---o "master"


A, B and C are topic branches.

 * A has one fix since it was merged up to "next".

 * B has finished.  It has been fully merged up to "master" and "next",
   and is ready to be deleted.

 * C has not merged to "next" at all.

We would want to allow C to be rebased, refuse A, and encourage
B to be deleted.

To compute (1):

	git rev-list ^master ^topic next
	git rev-list ^master        next

	if these match, topic has not merged in next at all.

To compute (2):

	git rev-list master..topic

	if this is empty, it is fully merged to "master".

DOC_END

#!/bin/sh

# An example hook script to update a checked-out tree on a git push.
#
# This hook is invoked by git-receive-pack(1) when it reacts to git
# push and updates reference(s) in its repository, and when the push
# tries to update the branch that is currently checked out and the
# receive.denyCurrentBranch configuration variable is set to
# updateInstead.
#
# By default, such a push is refused if the working tree and the index
# of the remote repository has any difference from the currently
# checked out commit; when both the working tree and the index match
# the current commit, they are updated to match the newly pushed tip
# of the branch. This hook is to be used to override the default
# behaviour; however the code below reimplements the default behaviour
# as a starting point for convenient modification.
#
# The hook receives the commit with which the tip of the current
# branch is going to be updated:
commit=$1

# It can exit with a non-zero status to refuse the push (when it does
# so, it must not modify the index or the working tree).
die () {
	echo >&2 "$*"
	exit 1
}

# Or it can make any necessary changes to the working tree and to the
# index to bring them to the desired state when the tip of the current
# branch is updated to the new commit, and exit with a zero status.
#
# For example, the hook can simply run git read-tree -u -m HEAD "$1"
# in order to emulate git fetch that is run in the reverse direction
# with git push, as the two-tree form of git read-tree -u -m is
# essentially the same as git switch or git checkout that switches
# branches while keeping the local changes in the working tree that do
# not interfere with the difference between the branches.

# The below is a more-or-less exact translation to shell of the C code
# for the default behaviour for git's push-to-checkout hook defined in
# the push_to_deploy() function in builtin/receive-pack.c.
#
# Note that the hook will be executed from the repository directory,
# not from the working tree, so if you want to perform operations on
# the working tree, you will have to adapt your code accordingly, e.g.
# by adding "cd .." or using relative paths.

if ! git update-index -q --ignore-submodules --refresh
then
	die "Up-to-date check failed"
fi

if ! git diff-files --quiet --ignore-submodules --
then
	die "Working directory has unstaged changes"
fi

# This is a rough translation of:
#
#   head_has_history() ? "HEAD" : EMPTY_TREE_SHA1_HEX
if git cat-file -e HEAD 2>/dev/null
then
	head=HEAD
else
	head=$(git hash-object -t tree --stdin </dev/null)
fi

if ! git diff-index --quiet --cached --ignore-submodules $head --
then
	die "Working directory has staged changes"
fi

if ! git read-tree -u -m "$commit"
then
	die "Could not update working tree to new HEAD"
fi

#!/bin/sh
#
# An example hook script to block unannotated tags from entering.
# Called by "git receive-pack" with arguments: refname sha1-old sha1-new
#
# To enable this hook, rename this file to "update".
#
# Config
# ------
# hooks.allowunannotated
#   This boolean sets whether unannotated tags will be allowed into the
#   repository.  By default they won't be.
# hooks.allowdeletetag
#   This boolean sets whether deleting tags will be allowed in the
#   repository.  By default they won't be.
# hooks.allowmodifytag
#   This boolean sets whether a tag may be modified after creation. By default
#   it won't be.
# hooks.allowdeletebranch
#   This boolean sets whether deleting branches will be allowed in the
#   repository.  By default they won't be.
# hooks.denycreatebranch
#   This boolean sets whether remotely creating branches will be denied
#   in the repository.  By default this is allowed.
#

# --- Command line
refname="$1"
oldrev="$2"
newrev="$3"

# --- Safety check
if [ -z "$GIT_DIR" ]; then
	echo "Don't run this script from the command line." >&2
	echo " (if you want, you could supply GIT_DIR then run" >&2
	echo "  $0 <ref> <oldrev> <newrev>)" >&2
	exit 1
fi

if [ -z "$refname" -o -z "$oldrev" -o -z "$newrev" ]; then
	echo "usage: $0 <ref> <oldrev> <newrev>" >&2
	exit 1
fi

# --- Config
allowunannotated=$(git config --type=bool hooks.allowunannotated)
allowdeletebranch=$(git config --type=bool hooks.allowdeletebranch)
denycreatebranch=$(git config --type=bool hooks.denycreatebranch)
allowdeletetag=$(git config --type=bool hooks.allowdeletetag)
allowmodifytag=$(git config --type=bool hooks.allowmodifytag)

# check for no description
projectdesc=$(sed -e '1q' "$GIT_DIR/description")
case "$projectdesc" in
"Unnamed repository"* | "")
	echo "*** Project description file hasn't been set" >&2
	exit 1
	;;
esac

# --- Check types
# if $newrev is 0000...0000, it's a commit to delete a ref.
zero=$(git hash-object --stdin </dev/null | tr '[0-9a-f]' '0')
if [ "$newrev" = "$zero" ]; then
	newrev_type=delete
else
	newrev_type=$(git cat-file -t $newrev)
fi

case "$refname","$newrev_type" in
	refs/tags/*,commit)
		# un-annotated tag
		short_refname=${refname##refs/tags/}
		if [ "$allowunannotated" != "true" ]; then
			echo "*** The un-annotated tag, $short_refname, is not allowed in this repository" >&2
			echo "*** Use 'git tag [ -a | -s ]' for tags you want to propagate." >&2
			exit 1
		fi
		;;
	refs/tags/*,delete)
		# delete tag
		if [ "$allowdeletetag" != "true" ]; then
			echo "*** Deleting a tag is not allowed in this repository" >&2
			exit 1
		fi
		;;
	refs/tags/*,tag)
		# annotated tag
		if [ "$allowmodifytag" != "true" ] && git rev-parse $refname > /dev/null 2>&1
		then
			echo "*** Tag '$refname' already exists." >&2
			echo "*** Modifying a tag is not allowed in this repository." >&2
			exit 1
		fi
		;;
	refs/heads/*,commit)
		# branch
		if [ "$oldrev" = "$zero" -a "$denycreatebranch" = "true" ]; then
			echo "*** Creating a branch is not allowed in this repository" >&2
			exit 1
		fi
		;;
	refs/heads/*,delete)
		# delete branch
		if [ "$allowdeletebranch" != "true" ]; then
			echo "*** Deleting a branch is not allowed in this repository" >&2
			exit 1
		fi
		;;
	refs/remotes/*,commit)
		# tracking branch
		;;
	refs/remotes/*,delete)
		# delete tracking branch
		if [ "$allowdeletebranch" != "true" ]; then
			echo "*** Deleting a tracking branch is not allowed in this repository" >&2
			exit 1
		fi
		;;
	*)
		# Anything else (is there anything else?)
		echo "*** Update hook: unknown type of update to ref $refname of type $newrev_type" >&2
		exit 1
		;;
esac

# --- Finished
exit 0

import aiomysql


class Database:
    def __init__(self):
        self.pool: aiomysql.Pool | None = None

    async def connect(self):
        if self.pool is not None:
            return
        self.pool = await aiomysql.create_pool(
            minsize=1,
            maxsize=10,
            host="127.0.0.1",
            port=3306,
            user="root",
            password="",
            db="vagacerta_db",
            autocommit=True,
            cursorclass=aiomysql.DictCursor,
        )
        print("[DB] ✅ Conectado ao MySQL (vagacerta_db) com sucesso.")

    async def close(self):
        if self.pool is not None:
            self.pool.close()
            await self.pool.wait_closed()
            self.pool = None

    async def execute(self, query: str, params: tuple = ()) -> int:
        async with self.pool.acquire() as conn:
            async with conn.cursor() as cursor:
                await cursor.execute(query, params)
                return cursor.lastrowid or cursor.rowcount

    async def fetch(self, query: str, params: tuple = (), one: bool = False):
        async with self.pool.acquire() as conn:
            async with conn.cursor() as cursor:
                await cursor.execute(query, params)
                if one:
                    return await cursor.fetchone()
                return await cursor.fetchall()


db = Database()

# MODELOS PYDANTIC (VALIDAÇÃO DO PAYLOAD DO EDITOR 2D)
# =========================================================
class SetorPayload(BaseModel):
    id_temp: int
    nome_setor: str
    prefixo: str
    cor_hex: str


class CelulaPreenchidaPayload(BaseModel):
    x: int
    y: int
    tipo: str  # 'ESTRADA' | 'ENTRADA' | 'SAIDA' | 'ACESSO' | 'FUNDO_SETOR' | 'VAGA'
    orientacao: str = "NENHUMA"
    setor_id_temp: Optional[int] = None
    identificador: Optional[str] = None
    livre: Optional[bool] = True
    mac_esp: Optional[str] = None


class AndarPayload(BaseModel):
    nome_andar: str
    grid_x: int
    grid_y: int
    setores: List[SetorPayload]
    celulas_preenchidas: List[CelulaPreenchidaPayload]


class EstabelecimentoSavePayload(BaseModel):
    nome: str
    id_usuario: int
    andares: List[AndarPayload]

# FUNÇÃO AUXILIAR DE PERSISTÊNCIA DO MAPA 2D EM TRANSAÇÃO
# =========================================================
async def gravar_andares_e_mapa(
    cursor, id_estabelecimento: int, andares: List[AndarPayload]
):
    for andar in andares:
        # 1. Insere o Andar
        await cursor.execute(
            """
            INSERT INTO andares (id_estabelecimento, nome_andar, grid_x, grid_y)
            VALUES (%s, %s, %s, %s)
            """,
            (id_estabelecimento, andar.nome_andar.strip(), andar.grid_x, andar.grid_y),
        )
        id_andar_banco = cursor.lastrowid

        # 2. Insere os Setores do andar e guarda o mapa { id_temp_do_vue : id_setor_do_mysql }
        mapa_setores_id = {}
        for setor in andar.setores:
            await cursor.execute(
                """
                INSERT INTO setores (id_andar, nome_setor, prefixo, cor_hex)
                VALUES (%s, %s, %s, %s)
                """,
                (
                    id_andar_banco,
                    setor.nome_setor.strip(),
                    setor.prefixo.strip().upper(),
                    setor.cor_hex,
                ),
            )
            mapa_setores_id[setor.id_temp] = cursor.lastrowid

        # 3. Cache local para não recriar o mesmo equipamento (id_setor + codigo_mac) várias vezes
        mapa_equipamentos_id = {}

        # 4. Percorre apenas os quadrados não-vazios enviados pelo Editor 2D
        for cel in andar.celulas_preenchidas:
            id_setor_real = (
                mapa_setores_id.get(cel.setor_id_temp) if cel.setor_id_temp else None
            )

            if cel.tipo == "VAGA":
                if not id_setor_real or not cel.mac_esp:
                    raise HTTPException(
                        status_code=400,
                        detail=f"A vaga na coordenada ({cel.x},{cel.y}) está sem setor ou sem MAC de equipamento.",
                    )
                chave_eq = (id_setor_real, cel.mac_esp.strip().upper())
                if chave_eq not in mapa_equipamentos_id:
                    await cursor.execute(
                        """
                        INSERT INTO equipamentos (id_setor, codigo_mac)
                        VALUES (%s, %s)
                        """,
                        (id_setor_real, cel.mac_esp.strip().upper()),
                    )
                    mapa_equipamentos_id[chave_eq] = cursor.lastrowid

                id_equip_real = mapa_equipamentos_id[chave_eq]

                await cursor.execute(
                    """
                    INSERT INTO vagas (id_equipamento, identificador, coord_x, coord_y, livre_para_estacionar)
                    VALUES (%s, %s, %s, %s, %s)
                    """,
                    (
                        id_equip_real,
                        (cel.identificador or "V1").strip().upper(),
                        cel.x,
                        cel.y,
                        1 if cel.livre else 0,
                    ),
                )
            else:
                # Elementos estruturais: ESTRADA, ENTRADA, SAIDA, ACESSO ou FUNDO_SETOR
                orientacao = (
                    cel.orientacao
                    if cel.orientacao in ("H", "V", "NENHUMA")
                    else "NENHUMA"
                )
                await cursor.execute(
                    """
                    INSERT INTO elementos_mapa (id_andar, id_setor, tipo, orientacao, coord_x, coord_y)
                    VALUES (%s, %s, %s, %s, %s, %s)
                    """,
                    (id_andar_banco, id_setor_real, cel.tipo, orientacao, cel.x, cel.y),
                )

@app.get("/estabelecimentos")
async def listar_estabelecimentos_completos():
    """Retorna todos os estabelecimentos com seus andares, setores, vagas e elementos do mapa."""
    estabelecimentos_rows = await db.fetch("""
        SELECT e.id_estabelecimento AS id, e.nome, e.inativo,
               COALESCE(ue.id_usuario, 1) AS id_usuario
        FROM estabelecimentos e
        LEFT JOIN usuario_estabelecimento ue ON e.id_estabelecimento = ue.id_estabelecimento
        ORDER BY e.id_estabelecimento ASC
        """)

    resultado = []
    for est in estabelecimentos_rows:
        id_est = est["id"]
        andares_rows = await db.fetch(
            """
            SELECT id_andar, nome_andar, grid_x, grid_y
            FROM andares
            WHERE id_estabelecimento = %s AND inativo = 0
            ORDER BY id_andar ASC
            """,
            (id_est,),
        )

        andares_formatados = []
        for andar in andares_rows:
            id_andar = andar["id_andar"]

            setores_rows = await db.fetch(
                """
                SELECT id_setor, nome_setor, prefixo, cor_hex, inativo AS inativo_offline
                FROM setores
                WHERE id_andar = %s
                ORDER BY id_setor ASC
                """,
                (id_andar,),
            )

            elementos_rows = await db.fetch(
                """
                SELECT id_elemento, id_setor, tipo, orientacao, coord_x, coord_y
                FROM elementos_mapa
                WHERE id_andar = %s
                """,
                (id_andar,),
            )

            vagas_rows = await db.fetch(
                """
                SELECT v.id_vaga, v.identificador, v.coord_x, v.coord_y, v.livre_para_estacionar,
                       eq.id_equipamento, eq.codigo_mac,
                       s.id_setor, s.nome_setor, s.prefixo, s.cor_hex
                FROM vagas v
                JOIN equipamentos eq ON v.id_equipamento = eq.id_equipamento
                JOIN setores s ON eq.id_setor = s.id_setor
                WHERE s.id_andar = %s AND v.inativo = 0
                """,
                (id_andar,),
            )

            andares_formatados.append(
                {
                    "id_andar": id_andar,
                    "nome_andar": andar["nome_andar"],
                    "grid_x": andar["grid_x"],
                    "grid_y": andar["grid_y"],
                    "setores": [
                        {
                            "id_setor": s["id_setor"],
                            "nome_setor": s["nome_setor"],
                            "prefixo": s["prefixo"],
                            "cor_hex": s["cor_hex"],
                            "inativo_offline": bool(s["inativo_offline"]),
                        }
                        for s in setores_rows
                    ],
                    "elementos_mapa": elementos_rows,
                    "vagas": [
                        {
                            "id_vaga": v["id_vaga"],
                            "identificador": v["identificador"],
                            "coord_x": v["coord_x"],
                            "coord_y": v["coord_y"],
                            "livre": bool(v["livre_para_estacionar"]),
                            "mac_esp": v["codigo_mac"],
                            "id_setor": v["id_setor"],
                            "nome_setor": v["nome_setor"],
                            "prefixo": v["prefixo"],
                            "cor_hex": v["cor_hex"],
                        }
                        for v in vagas_rows
                    ],
                }
            )

        resultado.append(
            {
                "id": id_est,
                "nome": est["nome"],
                "id_usuario": est["id_usuario"],
                "inativo": bool(est["inativo"]),
                "andares": andares_formatados,
            }
        )

    return resultado


@app.post("/estabelecimentos", status_code=status.HTTP_201_CREATED)
async def criar_estabelecimento(payload: EstabelecimentoSavePayload):
    async with db.pool.acquire() as conn:
        async with conn.cursor() as cursor:
            try:
                await conn.begin()

                await cursor.execute(
                    "INSERT INTO estabelecimentos (nome) VALUES (%s)",
                    (payload.nome.strip(),),
                )
                id_est = cursor.lastrowid

                await cursor.execute(
                    "INSERT INTO usuario_estabelecimento (id_usuario, id_estabelecimento) VALUES (%s, %s)",
                    (payload.id_usuario, id_est),
                )

                await gravar_andares_e_mapa(cursor, id_est, payload.andares)

                await conn.commit()
                return {"status": "sucesso", "id_estabelecimento": id_est}
            except Exception as e:
                await conn.rollback()
                raise HTTPException(
                    status_code=400, detail=f"Erro ao salvar no banco: {str(e)}"
                )


@app.put("/estabelecimentos/{id_estabelecimento}")
async def atualizar_estabelecimento(
    id_estabelecimento: int, payload: EstabelecimentoSavePayload
):
    async with db.pool.acquire() as conn:
        async with conn.cursor() as cursor:
            try:
                await conn.begin()

                await cursor.execute(
                    "UPDATE estabelecimentos SET nome = %s WHERE id_estabelecimento = %s",
                    (payload.nome.strip(), id_estabelecimento),
                )

                await cursor.execute(
                    "DELETE FROM usuario_estabelecimento WHERE id_estabelecimento = %s",
                    (id_estabelecimento,),
                )
                await cursor.execute(
                    "INSERT INTO usuario_estabelecimento (id_usuario, id_estabelecimento) VALUES (%s, %s)",
                    (payload.id_usuario, id_estabelecimento),
                )

                # Remove a estrutura anterior de andares (ON DELETE CASCADE limpa setores, vagas e elementos_mapa)
                # e grava a nova planta desenhada no Editor 2D
                await cursor.execute(
                    "DELETE FROM andares WHERE id_estabelecimento = %s",
                    (id_estabelecimento,),
                )

                await gravar_andares_e_mapa(cursor, id_estabelecimento, payload.andares)

                await conn.commit()
                return {"status": "sucesso", "id_estabelecimento": id_estabelecimento}
            except Exception as e:
                await conn.rollback()
                raise HTTPException(
                    status_code=400, detail=f"Erro ao atualizar no banco: {str(e)}"
                )


@app.patch("/estabelecimentos/{id_estabelecimento}/status")
async def alternar_inativo_estabelecimento(id_estabelecimento: int):
    est = await db.fetch(
        "SELECT inativo FROM estabelecimentos WHERE id_estabelecimento = %s",
        (id_estabelecimento,),
        one=True,
    )
    if not est:

    novo_status = 0 if est["inativo"] else 1
    await db.execute(
        """
        UPDATE estabelecimentos
        SET inativo = %s,
            data_inativacao = IF(%s = 1, CURRENT_TIMESTAMP(), NULL)
        WHERE id_estabelecimento = %s
        """,
        (novo_status, novo_status, id_estabelecimento),
    )
    return {"status": "sucesso", "inativo": bool(novo_status)}

def gerar_hash_senha(senha: str) -> str:
    salt = secrets.token_hex(16)
    derivado = hashlib.pbkdf2_hmac(
        "sha256", senha.encode("utf-8"), salt.encode("utf-8"), 100_000
    ).hex()
    return f"pbkdf2_sha256${salt}${derivado}"


def verificar_hash_senha(senha_digitada: str, hash_salvo: str) -> bool:
    if not hash_salvo or "$" not in hash_salvo:
        return False
    try:
        algoritmo, salt, derivado_salvo = hash_salvo.split("$", 2)
        if algoritmo != "pbkdf2_sha256":
            return False
        derivado_teste = hashlib.pbkdf2_hmac(
            "sha256", senha_digitada.encode("utf-8"), salt.encode("utf-8"), 100_000
        ).hex()
        return hmac.compare_digest(derivado_teste, derivado_salvo)
    except Exception:
        return False


def validar_regras_senha(senha: str, email: str):
    if len(senha) < 8:
        raise HTTPException(
            status_code=400, detail="A senha deve ter ao menos 8 caracteres."
        )
    if " " in senha:
        raise HTTPException(
            status_code=400, detail="A senha não pode conter espaços em branco."
        )
    if not re.search(r"[A-Z]", senha):
        raise HTTPException(
            status_code=400, detail="A senha deve conter ao menos uma letra maiúscula."
        )
    if not re.search(r"[a-z]", senha):
        raise HTTPException(
            status_code=400, detail="A senha deve conter ao menos uma letra minúscula."
        )
    if not re.search(r"\d", senha):
        raise HTTPException(
            status_code=400, detail="A senha deve conter ao menos um número."
        )
    if not re.search(r"[^A-Za-z0-9]", senha):
        raise HTTPException(
            status_code=400, detail="A senha deve conter ao menos um símbolo."
        )

    email_limpo = email.strip().lower()
    usuario_email = email_limpo.split("@")[0] if "@" in email_limpo else email_limpo
    senha_lower = senha.lower()

    if senha_lower == email_limpo or (
        len(usuario_email) >= 3 and usuario_email in senha_lower
    ):
        raise HTTPException(
            status_code=400,
            detail="A senha não pode conter nem ser parecida com o seu endereço de e-mail.",
        )


@app.post("/auth/cadastro", status_code=status.HTTP_201_CREATED)
async def cadastrar_usuario(payload: CadastroUsuarioPayload):
    nome = payload.nome.strip()
    email = payload.email.strip().lower()
    senha = payload.senha

    if len(nome) < 3:
        raise HTTPException(
            status_code=400, detail="Informe seu nome completo (mínimo 3 caracteres)."
        )
    if not re.match(r"^[^@\s]+@[^@\s]+\.[^@\s]+$", email):
        raise HTTPException(
            status_code=400, detail="Informe um endereço de e-mail válido."
        )

    validar_regras_senha(senha, email)

    existente = await db.fetch(
        "SELECT id_usuario FROM usuarios WHERE email = %s",
        (email,),
        one=True,
    )
    if existente:
        raise HTTPException(
            status_code=409, detail="Este e-mail já está cadastrado no sistema."
        )

    senha_hash = gerar_hash_senha(senha)
    novo_id = await db.execute(
        """
        INSERT INTO usuarios (nome, email, senha_hash, nivel_acesso)
        VALUES (%s, %s, %s, 'usuario')
        """,
        (nome, email, senha_hash),
    )

    return {
        "status": "sucesso",
        "usuario": {
            "id_usuario": novo_id,
            "nome": nome,
            "email": email,
            "nivel_acesso": "usuario",
        },
    }


@app.post("/auth/login")

    email = payload.email.strip().lower()

        )

    return {
        "status": "sucesso",

            "id_usuario": usuario["id_usuario"],
            "nome": usuario["nome"],
            "email": usuario["email"],

        },
    }

const panX = ref(0)
const panY = ref(0)
const zoomEscala = ref(1)

let toqueUnico = null
let pinchInicial = null

function distanciaEntreToques(t1, t2) {
  return Math.hypot(t2.clientX - t1.clientX, t2.clientY - t1.clientY)
}

function pontoMedioToques(t1, t2) {
  return {
    x: (t1.clientX + t2.clientX) / 2,
    y: (t1.clientY + t2.clientY) / 2
  }
}

function aoIniciarToqueGrid(e) {
  if (e.touches.length === 2) {
    toqueUnico = null
    const dist = distanciaEntreToques(e.touches[0], e.touches[1])
    const meio = pontoMedioToques(e.touches[0], e.touches[1])
    pinchInicial = {
      dist,
      escala: zoomEscala.value,
      panX: panX.value,
      panY: panY.value,
      meioX: meio.x,
      meioY: meio.y
    }
  } else if (e.touches.length === 1) {
    pinchInicial = null
    const t = e.touches[0]
    toqueUnico = {
      startX: t.clientX,
      startY: t.clientY,
      lastX: t.clientX,
      lastY: t.clientY,
      panXInicial: panX.value,
      panYInicial: panY.value,
      moveu: false
    }
  }
}

function aoMoverToqueGrid(e) {
  if (e.touches.length === 2 && pinchInicial) {
    const novaDist = distanciaEntreToques(e.touches[0], e.touches[1])
    const novoMeio = pontoMedioToques(e.touches[0], e.touches[1])
    const fator = novaDist / pinchInicial.dist
    zoomEscala.value = Math.min(2.5, Math.max(0.45, pinchInicial.escala * fator))
    panX.value = pinchInicial.panX + (novoMeio.x - pinchInicial.meioX)
    panY.value = pinchInicial.panY + (novoMeio.y - pinchInicial.meioY)
  } else if (e.touches.length === 1 && toqueUnico) {
    const t = e.touches[0]
    toqueUnico.lastX = t.clientX
    toqueUnico.lastY = t.clientY
    const dx = t.clientX - toqueUnico.startX
    const dy = t.clientY - toqueUnico.startY

    if (Math.hypot(dx, dy) > 7) {
      toqueUnico.moveu = true
      panX.value = toqueUnico.panXInicial + dx
      panY.value = toqueUnico.panYInicial + dy
    }
  }
}

function aoFinalizarToqueGrid(e) {
  if (e.touches.length < 2) {
    pinchInicial = null
  }
  if (e.touches.length === 0 && toqueUnico) {
    if (!toqueUnico.moveu) {
      const el = document.elementFromPoint(toqueUnico.lastX, toqueUnico.lastY)
      const celulaEl = el?.closest('[data-coord]')
      if (celulaEl) {

}

const listaAvisosValidacao = computed(() => {
  const avisos = []
  if (!formEstabelecimento.value?.andares) return avisos

  let temEntradaGlobal = false
  let temSaidaGlobal = false

  formEstabelecimento.value.andares.forEach((andar) => {
    const celulas = Object.values(andar.celulas || {})

    const vagasAndar = celulas.filter(c => c.tipo === 'VAGA')
    const semEquip = vagasAndar.some(c => !c.macEsp)
    const semSetor = vagasAndar.some(c => !c.setorId)
    const temEntrada = celulas.some(c => c.tipo === 'ENTRADA')
    const temSaida = celulas.some(c => c.tipo === 'SAIDA')
    const temAcesso = celulas.some(c => c.tipo === 'ACESSO')

    if (temEntrada) temEntradaGlobal = true
    if (temSaida) temSaidaGlobal = true

    if (vagasAndar.length === 0) {
      avisos.push(`[${andar.nome_andar}] Precisa ter ao menos 1 vaga cadastrada.`)
    }
    if (semEquip) {
      avisos.push(`[${andar.nome_andar}] Há vagas amarelas sem equipamento.`)
    }
    if (semSetor) {
      avisos.push(`[${andar.nome_andar}] Há vagas fora de um setor.`)
    }
    if (!temEntrada && !temSaida && !temAcesso) {
      avisos.push(`[${andar.nome_andar}] Precisa ter Entrada, Saída ou Acesso.`)
    }

    ;(andar.setores || []).forEach((setor) => {
      const vagasNoSetor = vagasAndar.filter(v => v.setorId === setor.id_setor).length
      if (vagasNoSetor === 0) {
        avisos.push(`[${andar.nome_andar}] Setor "${setor.nome_setor}" (${setor.prefixo}) precisa de 1 vaga.`)
      }
    })
  })

  if (!temEntradaGlobal) {
    avisos.push('[Geral] Obrigatório ter 1 ENTRADA em algum andar.')
  }
  if (!temSaidaGlobal) {
    avisos.push('[Geral] Obrigatório ter 1 SAÍDA em algum andar.')
  }

  return avisos
})

function obterProximoNumeroSetor(celulas, setorId, prefixo) {
  let maior = 0
  const regex = new RegExp(`^${prefixo}(\\d+)$`, 'i')
  Object.values(celulas).forEach((c) => {
    if (c.setorId === setorId && c.tipo === 'VAGA' && c.identificador) {
      const match = c.identificador.match(regex)
      if (match) {
        const num = parseInt(match[1], 10)
        if (num > maior) maior = num
      }
    }
  })
  return maior + 1
}

function numerarVagasDoSetor(celulas, setorId, prefixo) {
  const vagasDoSetor = Object.values(celulas)
    .filter(c => c.setorId === setorId && c.tipo === 'VAGA')
    .sort((a, b) => (a.y - b.y) || (a.x - b.x))

  vagasDoSetor.forEach((vaga, idx) => {
    vaga.identificador = `${prefixo}${idx + 1}`
  })
}

function aplicarTipoSelecao(novoTipo) {
  const celulas = andarEditorAtual.value.celulas
  const chaves = Array.from(selecionados.value)

  const temConflito = chaves.some((k) => {
    const atual = celulas[k]?.tipo
    return (novoTipo === 'ESTRADA' && atual === 'VAGA') || (novoTipo === 'VAGA' && atual === 'ESTRADA')
  })

  if (temConflito) {
    const msg = novoTipo === 'ESTRADA'
      ? 'Existem VAGAS na seleção. Deseja substituí-las por ESTRADAS?'
      : 'Existem ESTRADAS na seleção. Deseja substituí-las por VAGAS?'
    if (!confirm(msg)) return
  }

  salvarSnapshot()

  const chavesOrdenadas = chaves.sort((a, b) => {
    const [ax, ay] = a.split(',').map(Number)
    const [bx, by] = b.split(',').map(Number)
    return (ay - by) || (ax - bx)
  })

  chavesOrdenadas.forEach((k) => {
    const c = celulas[k]
    if (!c) return
    const jaEraVaga = c.tipo === 'VAGA'
    c.tipo = novoTipo
    c.orientacao = 'NENHUMA'

    if (novoTipo === 'VAGA') {
      c.macEsp = null
      c.livre = true
      if (c.setorId && c.prefixoSetor) {
        if (!jaEraVaga || !c.identificador) {
          const prox = obterProximoNumeroSetor(celulas, c.setorId, c.prefixoSetor)
          c.identificador = `${c.prefixoSetor}${prox}`
        }
      } else {
        c.identificador = ''
      }
    } else {
      c.macEsp = null
      c.identificador = ''
    }
  })

  mostrarMenuFlutuante.value = false
  selecionados.value.clear()
}

function removerTipoSelecao(tipoAlvo) {
  salvarSnapshot()
  const celulas = andarEditorAtual.value.celulas
  selecionados.value.forEach((k) => {
    const c = celulas[k]

    if (tipoAlvo === 'SETOR') {

      c.corSetor = null
      c.nomeSetor = null

    } else if (c.tipo === tipoAlvo) {
      c.tipo = 'VAZIO'
      c.macEsp = null
      c.identificador = ''
    }
  })
  limparSetoresOrfaosNoAndar(andarEditorAtual.value)
  mostrarMenuFlutuante.value = false
  selecionados.value.clear()
}

    alert('Digite um nome para o setor!')

  }
  if (!prefixoFinal) {
    alert('Digite um prefixo (1 a 3 letras) para o setor!')
    return
  }

  for (const andar of formEstabelecimento.value.andares) {
    limparSetoresOrfaosNoAndar(andar)
    for (const s of andar.setores || []) {

    nome_setor: nomeLimpo,

      c.nomeSetor = nomeLimpo

function aplicarCarimboNoGrid(startX, startY) {
  const carimbo = ferramentaCarimbo.value
  if (!carimbo) return

  const coordsAfetadas = []
  for (let cy = startY; cy < startY + carimbo.h; cy++) {
    for (let cx = startX; cx < startX + carimbo.w; cx++) {
      if (cx <= andarEditorAtual.value.grid_x && cy <= andarEditorAtual.value.grid_y) {
        coordsAfetadas.push(`${cx},${cy}`)
      }
    }
  }

  const celulas = andarEditorAtual.value.celulas

  if (carimbo.categoria === 'ACESSO') {
    const temVagaOuEstrada = coordsAfetadas.some(
      k => celulas[k].tipo === 'VAGA' || celulas[k].tipo === 'ESTRADA'
    )
    if (temVagaOuEstrada) {
      if (!confirm('Ao clicar em Ok, as vagas e as estradas da seleção serão removidas')) {
        return
      }
    }
    salvarSnapshot()
    coordsAfetadas.forEach((k) => {
      celulas[k].tipo = carimbo.tipo
      celulas[k].orientacao = carimbo.orientacao
      celulas[k].macEsp = null
      celulas[k].identificador = ''
    })
    ferramentaCarimbo.value = null
    return
  }

  if (carimbo.categoria === 'EQUIPAMENTO') {
    const vagasDebaixo = coordsAfetadas.filter(k => celulas[k].tipo === 'VAGA')
    if (vagasDebaixo.length === 0) {
      alert('Nenhuma VAGA encontrada sob este carimbo de equipamento! Crie as vagas primeiro.')
      return
    }
    coordenadasCarimboPendente.value = vagasDebaixo
    modalMacAberto.value = true
  }
}

function confirmarVinculoEquipamento() {
  if (!inputMacEsp.value.trim()) {
    alert('Informe o código MAC ou identificador do ESP32.')
    return
  }
  salvarSnapshot()
  const celulas = andarEditorAtual.value.celulas

  coordenadasCarimboPendente.value.forEach((k) => {
    const c = celulas[k]
    c.macEsp = inputMacEsp.value.trim().toUpperCase()
    if (c.setorId && c.prefixoSetor && !c.identificador) {
      const prox = obterProximoNumeroSetor(celulas, c.setorId, c.prefixoSetor)
      c.identificador = `${c.prefixoSetor}${prox}`
    }
  })

  modalMacAberto.value = false
  ferramentaCarimbo.value = null
}

async function finalizarSalvarEstacionamento() {
  if (!podeSalvarEstacionamento.value) {
    mostrarPainelAvisos.value = true
    return
  }
  if (salvandoNoBanco.value) return

  salvandoNoBanco.value = true
  try {
    await salvarEstacionamentoNoBanco()
    modalSucessoSalvarAberto.value = true
  } catch (err) {
    alert(`Erro ao salvar: ${err.message}`)
  } finally {
    salvandoNoBanco.value = false
  }
}

function confirmarModalSucessoSalvar() {
  modalSucessoSalvarAberto.value = false
  router.push({ name: 'admin-lista' })
}

function tratarTeclaGlobal(event) {
  if (event.key === 'Escape') {
    if (ferramentaCarimbo.value) ferramentaCarimbo.value = null
    if (mostrarMenuFlutuante.value) {
      mostrarMenuFlutuante.value = false
      selecionados.value.clear()
    }
    mostrarPainelAvisos.value = false
    menuTopoMobileAberto.value = false
    menuAdicionarMobileAberto.value = false
    dropdownSetoresAberto.value = false
  }
}

onMounted(() => {
  window.addEventListener('mouseup', finalizarSelecaoMouse)
  window.addEventListener('keydown', tratarTeclaGlobal)
})

onUnmounted(() => {
  window.removeEventListener('mouseup', finalizarSelecaoMouse)
  window.removeEventListener('keydown', tratarTeclaGlobal)
})

<script setup>
import { ref, computed } from 'vue'
import { useRouter } from 'vue-router'
import { carregarDadosDoBanco } from '../store/estacionamentoStore'
import {
  IconUserPlus,
  IconUser,
  IconMail,
  IconLock,
  IconEye,
  IconEyeOff,
  IconSquareRoundedCheck,
  IconAlertSquareRounded
} from '../icons'

const API_URL = 'http://127.0.0.1:8000'
const router = useRouter()

const nome = ref('')
const email = ref('')
const senha = ref('')
const confirmarSenha = ref('')

const mostrarSenha = ref(false)
const mostrarConfirmarSenha = ref(false)
const carregando = ref(false)
const erroMensagem = ref('')
const cadastroConcluido = ref(false)

// Aviso dinâmico em tempo real estilo Gmail (mostra apenas o que falta na senha)
const avisoFaltandoSenha = computed(() => {
  const s = senha.value
  if (!s) return ''

  if (/\s/.test(s)) {
    return 'A senha não pode conter espaços em branco.'
  }

  const faltando = []
  if (s.length < 8) faltando.push('mínimo de 8 caracteres')
  if (!/[A-Z]/.test(s)) faltando.push('1 letra maiúscula')
  if (!/[a-z]/.test(s)) faltando.push('1 letra minúscula')
  if (!/\d/.test(s)) faltando.push('1 número')
  if (!/[^A-Za-z0-9\s]/.test(s)) faltando.push('1 símbolo (!@#$...)')

  if (faltando.length === 0) return ''
  return `Falta na senha: ${faltando.join(', ')}.`
})

const avisoConfirmacaoSenha = computed(() => {
  if (!confirmarSenha.value) return ''
  if (senha.value !== confirmarSenha.value) {
    return 'As senhas não coincidem.'
  }
  return ''
})

// Permite clicar no botão assim que os campos básicos e a força da senha estiverem preenchidos
const formularioValido = computed(() => {
  return (
    nome.value.trim().length >= 3 &&
    /^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(email.value.trim()) &&
    senha.value.length >= 8 &&
    avisoFaltandoSenha.value === '' &&
    senha.value === confirmarSenha.value
  )
})

function senhaParecidaComEmail(senhaStr, emailStr) {
  const e = emailStr.trim().toLowerCase()
  const prefixoEmail = e.includes('@') ? e.split('@')[0] : e
  const sLower = senhaStr.toLowerCase()
  return sLower === e || (prefixoEmail.length >= 3 && sLower.includes(prefixoEmail))
}

async function realizarCadastro() {
  if (carregando.value || !formularioValido.value) return
  erroMensagem.value = ''

  // Regra estilo Gmail: só verifica semelhança com o e-mail após clicar em "Criar Conta"
  if (senhaParecidaComEmail(senha.value, email.value)) {
    erroMensagem.value = 'Por segurança, a senha não pode conter nem ser parecida com o seu e-mail.'
    return
  }

  carregando.value = true
  try {
    const resposta = await fetch(`${API_URL}/auth/cadastro`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        nome: nome.value.trim(),
        email: email.value.trim(),
        senha: senha.value
      })
    })

    const dados = await resposta.json().catch(() => ({}))
    if (!resposta.ok) {
      throw new Error(dados.detail || 'Não foi possível concluir o cadastro.')
    }

    await carregarDadosDoBanco()
    cadastroConcluido.value = true
  } catch (err) {
    erroMensagem.value = err.message
  } finally {
    carregando.value = false
  }
}
</script>

<script setup>
import { ref } from 'vue'
import { useRouter } from 'vue-router'
import { usuarioLogado } from '../store/estacionamentoStore'
import {
  IconLogin,
  IconMail,
  IconLock,
  IconEye,
  IconEyeOff,
  IconAlertSquareRounded
} from '../icons'

const API_URL = 'http://127.0.0.1:8000'
const router = useRouter()

const email = ref('')
const senha = ref('')
const mostrarSenha = ref(false)
const carregando = ref(false)
const erroMensagem = ref('')

async function realizarLogin() {
  if (carregando.value) return
  erroMensagem.value = ''

  if (!email.value.trim() || !senha.value) {
    erroMensagem.value = 'Preencha seu e-mail e sua senha para entrar.'
    return
  }

  carregando.value = true
  try {
    const resposta = await fetch(`${API_URL}/auth/login`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        email: email.value.trim(),
        senha: senha.value
      })
    })

    const dados = await resposta.json().catch(() => ({}))
    if (!resposta.ok) {
      throw new Error(dados.detail || 'Não foi possível realizar o login.')
    }

    if (usuarioLogado) {
      usuarioLogado.value = dados.usuario
    }
    localStorage.setItem('vagacerta_usuario', JSON.stringify(dados.usuario))
    const nivel = dados.usuario?.nivel_acesso
    if (nivel === 'admin_dev') {
      router.push({ name: 'admin-lista' })
    } else if (nivel === 'admin_estabelecimento') {
      router.push({ name: 'gerente' })
    } else {
      router.push({ name: 'motorista-home' })
    }
  } catch (err) {
    erroMensagem.value = err.message || 'Erro inesperado ao realizar o login.'
  } finally {
    carregando.value = false
  }
}
</script>

<template>
  <main class="tela-inicial">
    <div class="auth-card">
      <div class="auth-cabecalho">
        <div class="auth-icone-topo">
          <IconLogin :size="32" />
        </div>
        <h1>Bem-vindo de volta</h1>
        <p class="subtitulo">Acesse sua conta para gerenciar estacionamentos</p>
      </div>

      <Transition name="fade-pop">
        <div v-if="erroMensagem" class="alerta-auth erro item-icone-texto">
          <IconAlertSquareRounded :size="18" />
          <span>{{ erroMensagem }}</span>
        </div>
      </Transition>

      <form class="auth-form" @submit.prevent="realizarLogin">
        <div class="campo-grupo">
          <label for="login-email">E-mail</label>
          <div class="input-icone-container">
            <IconMail :size="18" class="icone-interno-esquerda" />
            <input
              id="login-email"
              v-model="email"
              type="email"
              placeholder="seu@email.com"
              autocomplete="email"
              :disabled="carregando"
              required
            />
          </div>
        </div>

        <div class="campo-grupo">
          <label for="login-senha">Senha</label>
          <div class="input-icone-container">
            <IconLock :size="18" class="icone-interno-esquerda" />
            <input
              id="login-senha"
              v-model="senha"
              :type="mostrarSenha ? 'text' : 'password'"
              placeholder="Digite sua senha"
              autocomplete="current-password"
              :disabled="carregando"
              required
            />
            <button
              type="button"
              class="btn-toggle-senha"
              :title="mostrarSenha ? 'Ocultar senha' : 'Mostrar senha'"
              @click="mostrarSenha = !mostrarSenha"
            >
              <IconEyeOff v-if="mostrarSenha" :size="19" />
              <IconEye v-else :size="19" />
            </button>
          </div>
        </div>

        <button
          type="submit"
          class="btn-primario btn-auth-submit item-icone-texto"
          :disabled="carregando"
        >
          <span v-if="carregando" class="spinner-btn"></span>
          <IconLogin v-else :size="20" />
          <span>{{ carregando ? 'Autenticando...' : 'Entrar' }}</span>
        </button>
      </form>

      <div class="auth-rodape">
        <span>Ainda não tem uma conta?</span>
        <router-link :to="{ name: 'cadastro' }" class="link-auth">
          Cadastre-se
        </router-link>
      </div>
    </div>
  </main>
</template>
